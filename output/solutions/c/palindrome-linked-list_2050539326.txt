/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     struct ListNode *next;
 * };
 */
struct ListNode * reverselist(struct ListNode* head) {
    if (head == NULL || head->next == NULL)
        return head;
    struct ListNode dummy;
    dummy.next = head;
    struct ListNode *prev = NULL, *curr = head, *next;
    while (curr != NULL) {
        next = curr->next;
        curr->next = prev;
        prev = curr;
        curr = next;
    }
    dummy.next = prev;
    return dummy.next;
}
bool isPalindrome(struct ListNode* head) {
    struct ListNode *slow = head, *fast = head;
    while (fast && fast->next) {
        slow = slow->next;
        fast = fast->next->next;
    }
    if (fast != NULL)
    slow = slow->next;   
    struct ListNode *head2 = reverselist(slow);
    struct ListNode *ptr1 = head, *ptr2 = head2;
    while (ptr2) {
        if (ptr1->val != ptr2->val)
            return false;
        ptr1 = ptr1->next;
        ptr2 = ptr2->next;
    }
    return true;
}
