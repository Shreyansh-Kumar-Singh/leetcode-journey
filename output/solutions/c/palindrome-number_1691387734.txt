bool isPalindrome(int x) {
    if (x < 0)
        return false;
    int rx = 0, digit, n = x;
    while (n != 0) {
        digit = n % 10;

        if (rx > INT_MAX / 10 || (rx == INT_MAX / 10 && digit > 7) ||
            rx < INT_MIN / 10 || (rx == INT_MIN / 10 && digit < -8))
            return false;

        rx = rx * 10 + digit;
        n /= 10;
    }
    if (rx == x)
        return true;
    else
        return false;
}
