/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     struct ListNode *next;
 * };
 */

typedef struct ListNode* N;
struct ListNode* removeNthFromEnd(struct ListNode* head, int n) {
    N ptr1 = head, ptr2 = head;
    for (int i = 0; i < n; i++) {
        ptr2 = ptr2->next;
    }
    while (ptr2 != NULL) {
        ptr1 = ptr1->next;
        ptr2 = ptr2->next;
    }
    if (ptr1 == head) {
        return head->next;
    } else {
        N ptr = head;
        while (ptr->next != ptr1) {
            ptr = ptr->next;
        }
        if (ptr1->next == NULL) {
            ptr->next = NULL;
            return head;
        }
        ptr->next = ptr1->next;
        return head;
    }
}
