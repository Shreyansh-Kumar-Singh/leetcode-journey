class Solution {
public:
    string addBinary(string a, string b) {
        reverse(a.begin() , a.end());
        reverse(b.begin() , b.end());
        string ans = "";
        int i = 0 , carry = 0;
        while(i<a.size() && i < b.size()){
            int sum = a[i] - '0' + b[i] -'0' + carry;
            ans+= (sum % 2) + '0';
            carry = sum / 2;
            i++;
        }
        string x = (a.size() > b.size())?a:b;
        while(i < x.size()){
            int sum = x[i]  - '0' + carry;
            ans+= (sum % 2) + '0';
            carry = sum /2;
            i++;
        }
        if(carry) ans+=(carry %2) + '0';
        reverse(ans.begin() , ans.end());
        return ans;
    }
};
