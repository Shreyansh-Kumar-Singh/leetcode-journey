class Solution {
public:
    string addBinary(string a, string b) {
        reverse(a.begin(), a.end());
        reverse(b.begin(), b.end());
        string ans = "";
        int i = 0, carry = 0;
        while (i < a.size() || i < b.size() || carry) {
            int sum = 0;
            if (i < a.size())
                sum += a[i] - '0';
            if (i < b.size())
                sum += b[i] - '0';
            if (carry)
                sum += carry;
            ans += (sum % 2) + '0';
            carry = sum / 2;
            i++;
        }
        reverse(ans.begin(), ans.end());
        return ans;
    }
};
