class Solution {
public:
    bool backspaceCompare(string s, string t) {
        int l = s.size() - 1, r = t.size() - 1;
        while (l >= 0 || r >= 0) {
            int count = 0;
            while (l >= 0) {
                if (s[l] == '#') {
                    count++;
                    l--;
                } else if (count > 0) {
                    l--;
                    count--;
                } else
                    break;
            }
            count = 0;
            while (r >= 0) {
                if (t[r] == '#') {
                    count++;
                    r--;
                } else if (count > 0) {
                    r--;
                    count--;
                } else
                    break;
            }
            if (l >= 0 && r >= 0) {
                if (s[l] != t[r])
                    return false;
            } else if (l >= 0 || r >= 0) {
                return false;
            }

            l--;
            r--;
        }
        return true;
    }
};
