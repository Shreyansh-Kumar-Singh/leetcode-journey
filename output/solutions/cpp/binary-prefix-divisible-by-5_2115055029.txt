class Solution {
public:
    vector<bool> prefixesDivBy5(vector<int>& nums) {
        vector<bool> ans;
        int x = 0;
        for(auto i : nums){
            x = ((x << 1)+ i) %5;
            if(x == 0) ans.push_back(true);
            else ans.push_back(false);
        }
        return ans;
    }
};
