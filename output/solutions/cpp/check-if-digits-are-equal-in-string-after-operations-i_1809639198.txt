class Solution {
public:
    bool hasSameDigits(string s) {
        while (s.size() > 2) {
            int x = 0;
            for (int i = 0; i < s.size() - 1; i++) {
                x = x * 10 + (((s[i] - '0') + (s[i + 1] - '0')) % 10);
            }
            s = to_string(x);
        }
        if (s.size() == 2 && s[0] == s[1])
            return true;
        else
            return false;
    }
};
