class Solution {
public:
    int coinChange(vector<int>& coins, int amount) {
        sort(coins.begin(), coins.end(), greater<int>());
        int i = 0, count = 0;
        while (amount > 0 && i < coins.size()) {
            if (coins[i] <= amount) {
                amount -= coins[i];
                count++;
            } else
                i++;
        }
        return amount == 0 ? count : -1;
    }
};
