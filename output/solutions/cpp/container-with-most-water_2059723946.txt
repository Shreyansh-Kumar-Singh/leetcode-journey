class Solution {
public:
    int maxArea(vector<int>& height) {
        int max1 = 0 ; 
        int l = 0 , r = height.size()-1;
        while(l < r){
            int min_h = min(height[l] , height[r]);
            max1 = max(min_h * (r-l), max1);
            if(height[l] < height[r]){
                l++;
            }else{
                r--;
            }
        }
        return max1;
    }
};
