class Solution {
public:
    vector<int> dailyTemperatures(vector<int>& temperatures) {
        int n = temperatures.size();
        if (n < 2)
            return {0};
        vector<int> ans;
        for (int i = 0; i < n; i++) {
            int a = 0;
            for (int j = i + 1; j < n; j++) {
                if (temperatures[i] < temperatures[j]) {
                    ans.push_back(j-i);
                    a++;
                    break;
                }
            }
            if (a == 0)
                ans.push_back(0);
        }
        return ans;
    }
};
