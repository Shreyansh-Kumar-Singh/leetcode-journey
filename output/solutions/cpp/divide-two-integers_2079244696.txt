class Solution {
public:
    int divide(int dividend, int divisor) {
        if(dividend == divisor) return 1;
        bool sign = true;
        if(dividend >= 0 && divisor <0) sign = false;
        if(dividend < 0 && divisor >0) sign = false;
        int ans= 0;
        long n = abs(dividend);
        long d = abs(divisor);
        while(n >= d){
            int cnt = 0;
            while(n >= (d << cnt+1)) cnt++;
            ans+=(1<<cnt);
            n -= (d<<cnt);
        }
        if(ans == (1<<31)){
            if(sign == true) return INT_MAX;
            else return INT_MIN;
        }
        return sign ? ans: (-1*ans);
    }
};
