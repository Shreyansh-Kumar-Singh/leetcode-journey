class Solution {
public:
    int findDuplicate(vector<int>& nums) {
        int n = nums.size();
        if(n == 1) return nums[0];
        sort(nums.begin() , nums.end());
        for(int i = 1 ; i < n ; i++ )
        {
            if(nums[i-1] == nums[i]) return nums[i-1];
        }
        return -1;
    }
};
