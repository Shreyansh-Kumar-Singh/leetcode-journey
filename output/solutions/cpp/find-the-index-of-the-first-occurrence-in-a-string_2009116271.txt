class Solution {
public:
    int strStr(string haystack, string needle) {
        int p = 0 , q = 0;
        while( p < haystack.size()){
            int curr = p;
            while(haystack[curr] == needle[q]){
                q++;
                curr++;
            }
            if(curr == needle.size() - 1) return p;
            p++;
        }
        return -1;
    }
};
