#include<unordered_set>
class Solution {
public:
    bool isHappy(int n) {
        unordered_set<int> seen;
        while(n != 1 && seen.find(n) == seen.end()){
            seen.insert(n);
            string s = to_string(n);
            long long  sum = 0;
            for(auto i: s){
                sum += (i - '0') * (i - '0');
            }
            n = sum;
        }
        return n == 1;
    }
}
