class Solution {
public:
    bool isLongPressedName(string name, string typed) {
        int p = 0;
        if(name[0] != typed[0] || name.size() != typed.size())return false;
        for(auto i : typed){
            if(i == name[p]){
                p++;
            }else if(i == name[p-1]){
                continue;
            }else{
                return false;
            }
        }
        return true;
    }
};
