class Solution {
public:
    int longestConsecutive(vector<int>& nums) {
        if(nums.size() < 2){
            return nums.size();
        }
        sort(nums.begin() , nums.end());
        int count = 0;
        int i = 1 ;
        int ans = 0;
        while(i < nums.size()){
            if(nums[i] == nums[i-1]){
                count++;
            }else{
                ans = max(ans,count);
                count = 0;
            }
            i++;
        }
        return ans;
    }
};
