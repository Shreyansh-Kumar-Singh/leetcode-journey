class Solution {
public:
    int lengthOfLongestSubstring(string s) {
        unordered_set<char> arr;
        int left = 0 , right  = 0 ;
        int n = s.length();
        int maxlen = 0;
        while(right < n){
            if(arr.find(s[right]) == arr.end()){
                arr.insert(s[right]);
                maxlen = max(maxlen , right-left+1);
                right++;
            }else{
                arr.erase(s[left]);
                left++;
            }
        }
        return maxlen;
    }
};
