class Solution {
public:
    int maxProduct(vector<int>& nums) {
        int ans = INT_MIN;
        for (int i = 0; i < nums.size(); i++) {
            long long mul = nums[i];
            ans = max(ans ,(int) mul);
            for (int j = i + 1; j < nums.size(); j++) {
                mul *= nums[j];
                ans = max(ans ,(int) mul);
            }
        }
        return ans;
    }
};
