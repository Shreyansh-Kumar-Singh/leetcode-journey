class Solution {
public:
    int getCommon(vector<int>& nums1, vector<int>& nums2) {
        sort(nums1.begin() , nums1.end());
        sort(nums2.begin() , nums2.end());
        int p1 = 0 , p2 = 0;
        int n = nums1.size() , m = nums2.size();
        while(p1 != n && p2 != m){
            if(nums1[p1] == nums2[p2]) return nums1[p1];
            else if(nums1[p1] < nums2[p2]) p1++;
            else p2++;
        }
        return -1;
    }
};
