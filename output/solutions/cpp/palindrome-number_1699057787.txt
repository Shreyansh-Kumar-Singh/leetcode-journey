class Solution {
public:
    bool isPalindrome(int x) {
        if (x < 0)
            return false;
        int n1 = x, rx = 0, digit = 0;
        while (n1 > 0) {
            digit = n1 % 10;
            n1 /= 10;
            rx = rx * 10 + digit;
        }
        if (rx == x)
            return true;
        else
            return false;
    }
};
