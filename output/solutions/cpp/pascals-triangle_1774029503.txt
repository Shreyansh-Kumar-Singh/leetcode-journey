class Solution {
public:
    vector<vector<int>> generate(int numRows) {
        vector<vector<int>> ans;
        for (int i = 1; i <= numRows; i++) {

            if (i == 1) {
                vector<int> arr;
                arr.push_back(1);
                ans.push_back(arr);
            } else {
                vector<int> arr(i, 0);
                int n = 1;
                for (int j = 0; j < i / 2 + 1; j++) {
                    if (j == 0) {
                        arr[j] = 1;
                        arr[i - j - 1] = 1;
                    } else {
                        n *= (i - j);
                        n /= j;
                        arr[j] = n;
                        arr[i - j - 1] = n;
                    }
                }
                ans.push_back(arr);
            }
        }
        return ans;
    }
};
