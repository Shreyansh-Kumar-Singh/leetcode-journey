class Solution {
public:
    double myPow(double x, int n) {
        long long N = n;
        double ans = 1;
        if(N < 0){
            x = 1/x;
            N *= -1;
        }
        while( N > 0){
            if( N % 2 == 0){
                x *= x;
                N /= 2;
            }else{
                ans *= x;
                N -= 1;
            }
        }
        return ans;
    }
};
