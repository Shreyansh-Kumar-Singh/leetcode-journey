class Solution {
public:
    vector<int> rearrangeArray(vector<int>& nums) {
        vector<int>pos , neg;
        for(auto i : nums)
        {
            if(i < 0 ) neg.push_back(i);
            else pos.push_back(i);
        }
        vector<int> ans = nums;
        for(int i = 0  ; i < pos.size() ; i++)
        {
            ans[2*i]  = pos[i];
            ans[2*i+1] = neg[i];
        }
        return ans;
    }
};
