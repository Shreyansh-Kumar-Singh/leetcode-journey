class Solution {
public:
    int removeDuplicates(vector<int>& nums) {
        int n = nums.size();
        int p1 = 0 , p2 = -1;
        int count = 0;
        while(p1 < n){
            if(p1 > 0 && nums[p1] == nums[p1-1])
                count++;
            else count = 0;
            if(count >= 2) p1++;
            else nums[++p2] = nums[p1++];
        }
        return p2+1;
    }
};
