class Solution {
public:
    int removeDuplicates(vector<int>& nums) {
        if (nums.size() < 2)
            return nums.size();
        set<int> st;
        for (int j = 0; j < nums.size(); j++) {
            st.insert(nums[j]);
        }
        int i = 0;
        for (auto k : st) {
            nums[i++] = k;
        }
        return i;
    }
};
