class Solution {
public:
    int reverse(int x) {
        string n = to_string(abs(x));

        std::reverse(n.begin() , n.end());

        long long ans = stoll(n);
        if (x < 0) {
            ans *= -1;
        }
        if (ans < INT_MIN || ans > INT_MAX)
            return 0;

        return (int)ans;
    }
};
