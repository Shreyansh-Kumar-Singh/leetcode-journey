class Solution {
public:
    string reverseOnlyLetters(string s) {
        int l = 0, r = s.size() - 1;
        string a = "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ";
        while (l < r) {
            if (l < r && a.find(s[l]) == string::npos)
                l++;
            if (l < r && a.find(s[r]) == string::npos)
                r--;
            swap(s[l], s[r]);
            l++;
            r--;
        }
        return s;
    }
};
