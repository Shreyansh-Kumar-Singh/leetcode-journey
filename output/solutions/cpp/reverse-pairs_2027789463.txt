class Solution {
public:
    int reversePairs(vector<int>& nums) {
        int count = 0;
        for(int i = 0 ; i < nums.size() ; i++){
            for(int j = i+1 ; j< nums.size() ; j++){
                long long a = (long long)2*nums[j];
                if(nums[i] > a)count++;
            }
        }
        return count;
    }
};
