class Solution {
public:
    bool is_vowel(char c) {
        c = tolower(c);
        if (c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u')
            return true;
        return false;
    }

    string reverseVowels(string s) {
        int l = 0, r = s.size() - 1;
        while (l < r) {
            while ( !is_vowel(s[l]) && l < s.size()/2){    l++;}
            while ( !is_vowel(s[r]) && r > s.size()/2){    r--;}
            swap(s[l] , s[r]);
            l++;
            r--;
        }
        return s;
    }
};
