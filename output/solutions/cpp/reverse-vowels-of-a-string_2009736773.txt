class Solution {
public:
    bool is_vowel(char c) {
        c = tolower(c);
        if (c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u')
            return true;
        return false;
    }

    string reverseVowels(string s) {
        int l = 0, r = s.size() - 1;

        while (l < r) {
            while (!is_vowel(s[l]) && l < r) {
                l++;
            }
            while (!is_vowel(s[r]) && l < r) {
                r--;
            }
            swap(s[l], s[r]);
            l++;
            r--;
        }
        return s;
    }
};
