class Solution {
public:
    int searchInsert(vector<int>& nums, int target) {
        auto n = find(nums.begin() , nums.end() , target);
        if( n != nums.end()) return n - nums.begin();
        for(int i = 0 ; i< nums.size() ; i++){
            if(nums[i] > target) return i;
        }
        return nums.size();
    }
};
