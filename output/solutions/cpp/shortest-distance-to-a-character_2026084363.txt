class Solution {
public:
    vector<int> shortestToChar(string s, char c) {
        vector<int> ans;
        int pos = INT_MAX;
        for (int i = 0; i < s.size(); i++) {
            if (s[i] == c) {
                pos = i;
            }
            if (pos != INT_MAX)
                ans.push_back(i - pos);
            else
                ans.push_back(pos);
        }
        pos = INT_MAX;
        for (int i = s.size() - 1; i > -1; i--) {
            if(s[i] == c) pos = i;
            if (pos != INT_MAX)
                ans[i] = min(ans[i], pos - i);
            else
                ans[i] = min(ans[i], pos);
        }
        return ans;
    }
};
