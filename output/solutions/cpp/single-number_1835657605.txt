class Solution {
public:
    int singleNumber(vector<int>& nums) {
        if(nums.size() < 2) return nums[0];
        sort(nums.begin() , nums.end());
        for(int i = 1; i< nums.size() ; i = i+2)
        {
            if(nums[i-1] != nums[i]) return nums[i-1];
        }
        return nums.back();
    }
};
