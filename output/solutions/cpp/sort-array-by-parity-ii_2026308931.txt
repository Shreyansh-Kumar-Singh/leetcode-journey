class Solution {
public:
    vector<int> sortArrayByParityII(vector<int>& nums) {
        int n = nums.size();
        int o = 1;
        int e = 0;
        vector<int> ans(n, -1);
        for (auto i : nums) {
            if (i % 2 == 0) {
                ans[e] = i;
                e += 2;
            } else {
                ans[o] = i;
                o += 2;
            }
        }
        return ans;
    }
};
