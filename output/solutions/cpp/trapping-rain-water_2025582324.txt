class Solution {
public:
    int trap(vector<int>& height) {
        int l = 0 , r = height.size()-1;
        int water = 0;
        int lmax = 0 , rmax = 0;
        while(l<r){
            if(height[l] <= height[r]){
                if(lmax > height[l]){
                    water = water + lmax - height[l];
                }else{
                    lmax = height[l];
                }
                l++;
            }else{
                if(rmax >= height[r]){
                    water = water + rmax - height[r];
                }else{
                    rmax = height[r];
                }
                r--;
            }
        }
        return water;
    }
};
