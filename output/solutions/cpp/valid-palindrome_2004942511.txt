class Solution {
public:
    bool isPalindrome(string s) {
        string ans;
        for(auto i: s){
            if(isalnum(i)) ans += tolower(i);
        }
        int l = 0;
        int r = ans.size()-1;
        while(l<r){
            if(ans[l] != ans[r]) return false;
            r--;
            l++; 
        }
        return true;
    }
};
