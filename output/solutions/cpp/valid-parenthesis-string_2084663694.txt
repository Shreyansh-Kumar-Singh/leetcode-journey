class Solution {
public:
    bool checkValidString(string s) {
        int low = 0, high = 0;
        for (char i : s) {
            if (i == '(') {
                low++;
                high++;
            } else if (i == ')') {
                low--;
                high--;
            } else {
                low--;
                high++;
            }
            if (high < 0)
                return false;
            low = max(low, 0);
        }
        return low == 0;
    }
};
