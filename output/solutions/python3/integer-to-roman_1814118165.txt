class Solution:
    def intToRoman(self, num: int) -> str:
        roman = {
            1: "I",
            4: "IV",
            5: "V",
            9: "IX",
            10: "X",
            40: "XL",
            50: "L",
            90: "XC",
            100: "C",
            400: "CD",
            500: "D",
            900: "CM",
            1000: "M",
        }
        l = list(roman.keys())
        l.reverse()
        ans = ""
        i = 0
        while num > 0:
            if num >= l[i] :
                ans += roman[l[i]]
                num -= l[i]
            else:
                i += 1
        return ans
