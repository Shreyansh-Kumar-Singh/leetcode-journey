class Solution:
    def intToRoman(self, num: int) -> str:
        d = {1:'I', 5:'V',10:'X',50:'L',100:'C',500:'D',1000:'M', 4:'IV', 9:'IX', 40:'XL', 90:'XC', 400:'CD', 900:'CM'}
        res = ''
        gtl = sorted(list(d.keys()))
        gtl.reverse()
        # print(f'gtl: {gtl}')
        i = 0
        while num > 0:
            # print(f'res: {res} | num: {num} | current gtl: {gtl[i]}')
            if(num >= gtl[i]):
                res += d[gtl[i]]
                num -= gtl[i]
            else:
                i+=1
        return res
