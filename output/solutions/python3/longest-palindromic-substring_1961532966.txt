class Solution:
    def palindrome(self , s: str) -> bool:
        return s == s[::-1]


    def longestPalindrome(self, s: str) -> str:
        maxlen = 0
        max1 = ""
        n = len(s)
        for i in range(n):
            for j in range(i , n):
                x = self.palindrome(s[i:j+1])
                if x:
                    if maxlen < (j-i+1) :
                        maxlen = j-i+1
                        max1 = s[i:j+1]
        return max1
