class Solution:
    def romanToInt(self, s: str) -> int:
        roman = {
            "I": 1,
            "V": 5,
            "X": 10,
            "L": 50,
            "C": 100,
            "D": 500,
            "M": 1000,
        }
        ans = 0
        prev = 0
        for i in reversed(s):
            value = roman[i]
            if(value < prev):
                ans -= value
            else:
                ans += value
                prev = value
        return ans
